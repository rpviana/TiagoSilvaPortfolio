// Componente auxiliar para o ProjectsSettings.tsx
// Cole este código ANTES da função ProjectPreview no arquivo ProjectsSettings.tsx

interface PageStylesEditorProps {
  siteContent: Record<string, any>;
  onContentChange: (key: string, field: 'valuePt' | 'valueEn', value: string) => void;
}

function PageStylesEditor({ siteContent, onContentChange }: PageStylesEditorProps) {
  const fields = [
    { key: "projects_title", label: "Título da Página", type: "text" },
    { key: "projects_description", label: "Descrição", type: "textarea" },
    { key: "projects_bg_color", label: "Cor de Fundo", type: "color" },
    { key: "projects_title_color", label: "Cor do Título", type: "color" },
    { key: "projects_collaborative_title", label: "Título Trabalho Colaborativo", type: "text" },
    { key: "projects_collaborative_text1", label: "Texto Colaborativo 1", type: "textarea" },
    { key: "projects_collaborative_text2", label: "Texto Colaborativo 2", type: "textarea" },
    { key: "projects_collaboration1", label: "Colaboração 1", type: "text" },
    { key: "projects_collaboration2", label: "Colaboração 2", type: "text" },
    { key: "projects_collaboration3", label: "Colaboração 3", type: "text" },
    { key: "projects_collaboration4", label: "Colaboração 4", type: "text" },
  ];

  return (
    <div className="space-y-6">
      <Tabs defaultValue="pt">
        <TabsList className="grid w-full grid-cols-2">
          <TabsTrigger value="pt">🇵🇹 Português</TabsTrigger>
          <TabsTrigger value="en">🇬🇧 English</TabsTrigger>
        </TabsList>

        <TabsContent value="pt" className="space-y-4 mt-4">
          {fields.map(field => (
            <div key={field.key}>
              <label className="block text-sm font-medium mb-2">
                {field.label}
              </label>
              {field.type === 'textarea' ? (
                <Textarea
                  value={siteContent[field.key]?.valuePt || ''}
                  onChange={(e) => onContentChange(field.key, 'valuePt', e.target.value)}
                  rows={3}
                />
              ) : field.type === 'color' ? (
                <Input
                  type="color"
                  value={siteContent[field.key]?.valuePt || '#ffffff'}
                  onChange={(e) => onContentChange(field.key, 'valuePt', e.target.value)}
                />
              ) : (
                <Input
                  value={siteContent[field.key]?.valuePt || ''}
                  onChange={(e) => onContentChange(field.key, 'valuePt', e.target.value)}
                />
              )}
            </div>
          ))}
        </TabsContent>

        <TabsContent value="en" className="space-y-4 mt-4">
          {fields.map(field => (
            <div key={field.key}>
              <label className="block text-sm font-medium mb-2">
                {field.label}
              </label>
              {field.type === 'textarea' ? (
                <Textarea
                  value={siteContent[field.key]?.valueEn || ''}
                  onChange={(e) => onContentChange(field.key, 'valueEn', e.target.value)}
                  rows={3}
                />
              ) : field.type === 'color' ? (
                <Input
                  type="color"
                  value={siteContent[field.key]?.valueEn || '#ffffff'}
                  onChange={(e) => onContentChange(field.key, 'valueEn', e.target.value)}
                />
              ) : (
                <Input
                  value={siteContent[field.key]?.valueEn || ''}
                  onChange={(e) => onContentChange(field.key, 'valueEn', e.target.value)}
                />
              )}
            </div>
          ))}
        </TabsContent>
      </Tabs>
    </div>
  );
}
